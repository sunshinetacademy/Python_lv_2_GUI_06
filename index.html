المستوى الأول 🟢 Listbox — الأساسيات

listbox_basics.py
import tkinter as tk

root = tk.Tk()
root.title("أول Listbox")
root.geometry("360x300")

lb = tk.Listbox(root, font=("Arial", 14), height=8)
lb.pack(fill="both", expand=True, padx=14, pady=14)

# ➕ إضافة عناصر في النهاية
lb.insert(tk.END, "تفاح 🍎")
lb.insert(tk.END, "موز 🍌")
lb.insert(tk.END, "برتقال 🍊")

# ➕ إضافة في مكان محدد (الفهرس 1)
lb.insert(1, "عنب 🍇")

# 📏 عدد العناصر
print("عدد العناصر:", lb.size())

# 📖 قراءة عنصر بالفهرس
print("الأول:", lb.get(0))

# 📖 قراءة كل العناصر
print("الكل:", lb.get(0, tk.END))

# 🗑️ حذف عنصر بالفهرس
lb.delete(0)

root.mainloop()
الدالةبتعمل إيه؟ملاحظة مهمة
insert(index, text)➕ يضيف عنصر عند الفهرسtk.END = في النهاية
delete(index)🗑️ يحذف عنصر واحدمش بيمسح الكل!
delete(0, tk.END)🧹 يمسح كل العناصرلازم حدّين للإفراغ الكامل
size()📏 عدد العناصربيرجع int
get(index)📖 نص عنصر واحدلازم فهرس — get() لوحدها = خطأ
get(0, tk.END)📖 كل العناصربيرجع tuple من النصوص
see(index)👁️ يمرّر لحد العنصريخليه ظاهر في الشاشة
⚠️ فخ مشهور: lb.delete(0) يحذف عنصر واحد بس! لو عايز تفضى القائمة كلها لازم lb.delete(0, tk.END). وlb.get() بدون فهرس بترمي TypeError.

المستوى الثاني 🟡 التحديد — curselection و selectmode

القوة الحقيقية للـ Listbox إن المستخدم يختار. وعشان نعرف اختار إيه، نستخدم curselection().

listbox_selection.py
import tkinter as tk
from tkinter import messagebox

root = tk.Tk()
root.title("التحديد في Listbox")
root.geometry("380x340")

# selectmode: browse | single | multiple | extended
lb = tk.Listbox(
    root,
    selectmode="extended",
    activestyle="underline",
    selectbackground="#a3e635",
    selectforeground="#0c2b22",
    font=("Arial", 13),
    height=8,
)
lb.pack(fill="both", expand=True, padx=14, pady=14)

for item in ["تفاح", "موز", "برتقال", "عنب", "مانجو", "فراولة"]:
    lb.insert(tk.END, item)


def show_selected():
    indices = lb.curselection()        # 👈 tuple من الفهارس
    if not indices:
        messagebox.showwarning("تنبيه", "اختار عنصر الأول!")
        return
    names = [lb.get(i) for i in indices]
    result.config(text="اخترت: " + "، ".join(names))


def delete_selected():
    indices = lb.curselection()
    # ⚠️ نحذف من الآخر للأول عشان الفهارس ما تتلخبطش
    for i in reversed(indices):
        lb.delete(i)
    result.config(text=f"تم حذف {len(indices)} عنصر")


btns = tk.Frame(root)
btns.pack(fill="x", padx=14, pady=(0, 8))
tk.Button(btns, text="👁️ اعرض المحدد", command=show_selected).pack(side="right", expand=True, fill="x", padx=3)
tk.Button(btns, text="🗑️ احذف المحدد", command=delete_selected).pack(side="right", expand=True, fill="x", padx=3)

result = tk.Label(root, text="اختار عنصر واضغط الزر", font=("Arial", 11))
result.pack(pady=(0, 12))

lb.selection_set(0)      # تحديد أول عنصر بالكود
lb.see(0)                # التأكد إنه ظاهر

root.mainloop()

🎛️ selectmode — أهم خاصية في المحاضرة

القيمةالمستخدم يقدر يختارطريقة التعاملمناسب لـ
"browse"عنصر واحد (الافتراضي)ضغط عادي، والسحب يغيّر الاختياراختار ملف تفتحه
"single"عنصر واحدضغط عادي فقطاختيار بسيط
"multiple"أي عددكل ضغطة تعكس العنصر (toggle)اختار هواياتك ☑️
"extended"أي عددCtrl+ضغط للإضافة، Shift+ضغط لنطاقسلة مشتريات 🛒
الدالةبتعمل إيه؟
curselection()🎯 tuple بالفهارس المحددة — مثال (0, 2, 3)
selection_set(i)✅ يحدد العنصر i بالكود
selection_clear(0, tk.END)❌ يشيل كل التحديد
selection_includes(i)❓ هل العنصر i محدد؟ True/False
activate(i)📍 يخلي العنصر i هو "النشط"
⚠️ أخطر فخ في الحذف المتعدد: لو حذفت من الأول للآخر، الفهارس بتتغير بعد كل حذف وهتحذف عناصر غلط! الحل دايماً: for i in reversed(indices): lb.delete(i) — من الآخر للأول. 🔁
💡 Remember: curselection() بترجع فهارس مش نصوص! عشان تجيب النص لازم lb.get(i). ودايماً افحص if not indices: قبل ما تستخدمها.

المستوى الثالث 🔴 Scrollbar — الربط في الاتجاهين ↕️

لما العناصر تزيد عن الـ heightفي اتجاهين — مش سطر واحد!

scrollbar_binding.py
import tkinter as tk

root = tk.Tk()
root.title("Listbox + Scrollbar")
root.geometry("380x320")

# 1️⃣ حاوية Frame تجمع القائمة والشريط جنب بعض
frame = tk.Frame(root)
frame.pack(fill="both", expand=True, padx=14, pady=14)

# 2️⃣ الشريط على اليمين ويملأ الطول
bar = tk.Scrollbar(frame, orient="vertical", width=16)
bar.pack(side="right", fill="y")

# 3️⃣ القائمة: الاتجاه الأول (القائمة تُبلغ الشريط)
lb = tk.Listbox(
    frame,
    height=8,
    font=("Arial", 13),
    yscrollcommand=bar.set,     # 👈 lb ➜ bar
)
lb.pack(side="left", fill="both", expand=True)

# 4️⃣ الاتجاه الثاني (الشريط يأمر القائمة)
bar.config(command=lb.yview)    # 👈 bar ➜ lb

for i in range(1, 41):
    lb.insert(tk.END, f"منتج رقم {i}")

root.mainloop()

المستوى الرابع 🟣 Treeview — الجدول بأعمدة 🌳

الـ Listbox عمود واحد. لو عايز اسم + كمية + سعر في صفوف مرتبة → ttk.Treeview!

treeview_demo.py
import tkinter as tk
from tkinter import ttk

root = tk.Tk()
root.title("Treeview — جدول المنتجات")
root.geometry("520x340")

frame = tk.Frame(root)
frame.pack(fill="both", expand=True, padx=14, pady=14)

bar = tk.Scrollbar(frame, orient="vertical")
bar.pack(side="right", fill="y")

cols = ("item", "qty", "price")
tv = ttk.Treeview(
    frame,
    columns=cols,
    show="headings",        # 👈 يخفي عمود الشجرة ويظهر العناوين فقط
    height=8,
    selectmode="extended",
    yscrollcommand=bar.set,
)

# عناوين الأعمدة
tv.heading("item", text="المنتج")
tv.heading("qty", text="الكمية")
tv.heading("price", text="السعر")

# عرض ومحاذاة كل عمود
tv.column("item", width=200, anchor="w")
tv.column("qty", width=80, anchor="center")
tv.column("price", width=100, anchor="center")

tv.pack(side="left", fill="both", expand=True)
bar.config(command=tv.yview)

# ➕ إدراج صف بقيم متعددة
rows = [
    ("تفاح 🍎", 6, 12),
    ("موز 🍌", 4, 8),
    ("برتقال 🍊", 3, 10),
    ("عنب 🍇", 2, 25),
]
for row in rows:
    tv.insert("", tk.END, values=row)


def show_row():
    for iid in tv.selection():
        values = tv.item(iid, "values")
        print("الصف المحدد:", values)


def remove_row():
    for iid in tv.selection():
        tv.delete(iid)


btns = tk.Frame(root)
btns.pack(fill="x", padx=14, pady=(0, 12))
tk.Button(btns, text="👁️ اعرض الصف", command=show_row).pack(side="right", expand=True, fill="x", padx=3)
tk.Button(btns, text="🗑️ احذف الصف", command=remove_row).pack(side="right", expand=True, fill="x", padx=3)

root.mainloop()
العنصرالوظيفة

tv.heading(id, text=...)النص الظاهر في رأس العمود
tv.column(id, width, anchor)عرض العمود والمحاذاة (w/e/center)
tv.insert("", tk.END, values=row)إدراج صف — values مش نص واحد!
tv.selection()tuple بمعرّفات الصفوف (iids) — مش فهارس!
tv.item(iid, "values")قيم الصف كـ tuple
tv.get_children()كل معرّفات الصفوف

⚠️ فرق جوهري: في Listbox التحديد بيرجع أرقام فهارس، وفي Treeview بيرجع معرّفات نصية (iids). عشان كده في Treeview الحذف المتعدد آمن مباشرة — مفيش مشكلة تغيّر الفهارس! ✅

def refresh_cart():
    """إعادة بناء Listbox السلة من قائمة cart"""
    cart_box.delete(0, tk.END)
    for it in cart:
        line = f"{it['name']}  ×{it['qty']}  =  {it['qty'] * it['price']} ج"
        cart_box.insert(tk.END, line)

    total = sum(it["qty"] * it["price"] for it in cart)
    count_label.configure(text=f"🧾 عدد الأصناف: {cart_box.size()}")
    total_label.configure(text=f"💰 الإجمالي: {total} جنيه")


def add_to_cart():
    selected = tv.selection()
    if not selected:
        messagebox.showwarning("تنبيه", "⚠️ حدّد منتج من الكتالوج الأول!")
        return

    try:
        qty = int(qty_entry.get())
        if qty <= 0:
            raise ValueError
    except ValueError:
        messagebox.showerror("خطأ", "❌ الكمية لازم رقم صحيح موجب!")
        return

    added = 0
    for iid in selected:
        name, price, stock = tv.item(iid, "values")
        price = int(price)
        stock = int(stock)

        already = sum(i["qty"] for i in cart if i["name"] == name)
        if already + qty > stock:
            messagebox.showerror(
                "المخزون",
                f"❌ {name}: المتاح {stock} فقط (في السلة {already})",
            )
            continue

        found = False
        for it in cart:
            if it["name"] == name:
                it["qty"] += qty
                found = True
                break
        if not found:
            cart.append({"name": name, "qty": qty, "price": price})
        added += 1

    if added:
        refresh_cart()
        cart_box.see(tk.END)          # مرّر لآخر عنصر أُضيف
        status.configure(text=f"✅ أُضيف {added} صنف للسلة", fg=LIME)


def remove_selected():
    indices = cart_box.curselection()
    if not indices:
        messagebox.showwarning("تنبيه", "⚠️ حدّد عنصر من السلة!")
        return
    # 🔁 من الآخر للأول حتى لا تتغير الفهارس
    for i in reversed(indices):
        cart.pop(i)
    refresh_cart()
    status.configure(text=f"🗑️ حُذف {len(indices)} صنف", fg=ROSE)


def clear_cart():
    if not cart:
        messagebox.showinfo("السلة", "🛒 السلة فارغة أصلاً!")
        return
    if messagebox.askyesno("تأكيد", "متأكد إنك عايز تفضّي السلة؟"):
        cart.clear()
        cart_box.delete(0, tk.END)     # الطريقة الصحيحة للإفراغ
        refresh_cart()
        status.configure(text="🧹 السلة اتفضّت", fg=MINT)


def checkout():
    if not cart:
        messagebox.showwarning("تنبيه", "🛒 السلة فارغة!")
        return
    total = sum(i["qty"] * i["price"] for i in cart)
    lines = [f"• {i['name']} ×{i['qty']} = {i['qty']*i['price']} ج" for i in cart]
    receipt = "\n".join(lines) + f"\n\n💰 الإجمالي: {total} جنيه"
    if messagebox.askyesno("إتمام الشراء", receipt + "\n\nتأكيد؟"):
        messagebox.showinfo("تم ✅", f"تم الشراء بمبلغ {total} جنيه")
        cart.clear()
        refresh_cart()
        status.configure(text="🎉 تم إتمام الشراء!", fg=LIME)


def sort_catalog(col):
    """فرز الكتالوج بالضغط على رأس العمود"""
    global sort_state
    idx = {"name": 0, "price": 1, "stock": 2}[col]
    reverse = sort_state.get(col, False)
    rows = sorted(CATALOG, key=lambda r: r[idx], reverse=reverse)
    sort_state[col] = not reverse
    for iid in tv.get_children():
        tv.delete(iid)
    for row in rows:
        tv.insert("", tk.END, values=row)
